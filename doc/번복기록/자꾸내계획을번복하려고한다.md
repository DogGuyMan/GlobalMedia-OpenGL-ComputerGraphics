            lastMat  = nullptr;
        }
        // ── 결정2a: material 이 *바뀔 때만* sampler 바인딩 ──
        if (material != lastMat) {                        // ← 직전 항목과 비교 = 배치
            BindSamplers(rc, material->Properties, *program);
            lastMat = material;
        }
        // ── 결정2b/3/4: 매 draw (model/ROP/draw) ──
        UploadMaterialUboMembers(*program, material->Properties);
        rc.ApplyRenderStateBlock(Pass::DefaultRenderStateBlockOf(material->GetPass()));
        program->UpdateUniformBlock("DrawBlock", &cmd.modelMatrix, sizeof(glm::mat4), 0);
        rc.BindVAO(mesh->GetVAO());
        rc.DrawIndexed(mesh->GetIndexCount());
    }
}
lastProg/lastMat 는 항목과 항목 사이에 살아남는 상태입니다. SortMultiStage(:113)가 같은 program/material 끼리 인접하게 모아두기 때문에, "직전과 같으면 skip" 이 곧 그룹당 1회 업로드가 됩니다.

→ 이 상태는 어떤 단일 메시의 것도 아니라 루프의 것 입니다. 이게 긴장의 뿌리입니다.

1. §0.5 가 그 상태를 못 드는 이유
마이그레이팅계획안.md §0.5:


void RenderableProcessor::Process(ICommandRecorder& rec, const Camera& cam) {
    for (auto* r : mItems) {
        rec.ApplyRenderStateBlock(r->GetRenderStateBlock());
        r->Render(rec, cam);     // ← 각 항목이 자기 자신을 전부 발행

이 루프엔 lastProg/lastMat 가 없습니다. 발행이 r->Render 안으로 들어갔으니, 한 메시는 옆 메시가 뭐였는지 모릅니다 → "직전과 같으면 skip" 을 할 주체가 사라집니다.

2. Option A — §0.5 그대로 (r->Render 자가발행)
A-naive (가장 충실, 배치 없음)
MeshRenderer 가 IRenderable 잎으로서 자기를 전부 그림:


// mesh_renderer.cpp (신규)
void MeshRenderer::Render(ICommandRecorder& rec, const Scene::Camera& cam) const
{
    if (!Material || !Mesh) return;
    const Program* program = Material->GetProgram();
    if (!program) return;

    rec.UseProgram(*program);
    // FrameBlock(view/proj) — cam 에서 매번 도출해 업로드 (★ program 그룹 배치 없음)
    const glm::mat4 view = cam.GetViewMatrix();
    const glm::mat4 proj = cam.GetProjectionMatrix();
    program->UpdateUniformBlock("FrameBlock", &view, sizeof(glm::mat4), 0);
    program->UpdateUniformBlock("FrameBlock", &proj, sizeof(glm::mat4), sizeof(glm::mat4));
    program->BindUniformBlocks();

    BindSamplers(rec, Material->Properties, *program);          // ★ material 그룹 배치 없음
    UploadMaterialUboMembers(*program, Material->Properties);

    const glm::mat4 model = GetOwner()->GetWorldMatrix();
    program->UpdateUniformBlock("DrawBlock", &model, sizeof(glm::mat4), 0);
    rec.BindVAO(Mesh->GetVAO());
    rec.DrawIndexed(Mesh->GetIndexCount());
}
인터페이스 LSP-3 깨끗: IRenderable 은 Render/QueueLayer/GetRenderStateBlock 만. processor 는 mesh 특수지식 0 (§0.5 의 "mesh 특수 0" 그대로).
잃는 것: program 100개가 같아도 FrameBlock 을 100번, material 같아도 sampler 를 100번 재바인딩.
픽셀: 동일 (같은 값을 더 자주 쓸 뿐). 성능만 이론상 회귀.
규모 현실: view/proj = 128바이트 UBO write + sampler bind 가 draw 당 몇 번. 이 게임 규모(draw 수십~수백)에선 거의 측정 불가 (YAGNI).
A-dedup (flat 유지 + 배치를 DeviceContext 로 흡수)
"배치 상태"를 루프가 아니라 DeviceContext(이미 GL state 단일 권위, D-RS-1) 가 들게 함. 잎 코드는 A-naive 와 동일하되, 중복 호출을 DeviceContext/Program 이 no-op:

3. Option B — processor 가 배치를 계속 구동
RenderableProcessor 가 현 Process 루프(lastProg/lastMat)를 그대로 유지하고, IRenderable 은 정렬·ROP·식별만 일반화. 발행은 processor 가:


void RenderableProcessor::Process(ICommandRecorder& rec, const Camera& cam)
{
    const Program*  lastProg = nullptr;
    const Material* lastMat  = nullptr;
    rec.InvalidateStateCache();
    const glm::mat4 view = cam.GetViewMatrix();
    const glm::mat4 proj = cam.GetProjectionMatrix();

    for (auto* r : mItems)               // r = IRenderable*
    {
        // ★ 배치하려면 program/material 을 알아야 함 → IRenderable LSP-3 엔 없음.
        //   → DrawCommand(meshRenderer 포인터)를 internal 로 유지하거나
        //     IRenderable 에 GetMaterial()/GetProgram() 추가(LSP-3 깨짐).
        const MeshRenderer* mr = static_cast<const MeshRenderer*>(r);  // 또는 DrawCommand 경유

        if (mr->Material != lastMat) { BindSamplers(...); lastMat = mr->Material; }
        rec.ApplyRenderStateBlock(r->GetRenderStateBlock());   // ← 여기만 IRenderable 다형 사용

회귀 위험 0: 현 코드와 거의 동일, 픽셀·성능 둘 다 보존.
대가: 배치하려면 processor 가 program/material 을 알아야 하는데 IRenderable 은 안 줌 → (B1) DrawCommand(meshRenderer 포인터)를 internal 구현 디테일로 유지(processor 가 mesh 지식 보유) 또는 (B2) IRenderable 에 GetMaterial() 추가(LSP-3·"mesh 특수 0" 위배). 즉 §0.5 의 "flat·mesh 특수 0" 표현과 어긋남 — processor 가 "두꺼워짐".
4. 비교 + 추천
flat(§0.5) 충실	LSP-3 / mesh특수0	배치 보존	회귀 위험	변경 범위
A-naive	◎ 완전	◎	✗ (per-draw 재업로드)	픽셀0 / 성능 미미	가장 작음
A-dedup	◎ (루프는 §0.5 그대로)	◎	◎ (멱등 dedup)	0	DeviceContext/Program 에 캐시 추가

추천: A-naive 로 착수 → 필요 시 A-dedup 로 무손실 진화.

근거:

마이그레이팅계획안 §0.5/D5 의 "Flat·mesh 특수 0" 의도에 가장 충실한 건 A 계열. B 는 processor 가 mesh 를 계속 알아야 해서 그 의도를 깬다.
A-naive 가 잃는 배치는 픽셀 무관·이 규모에선 측정 불가 → 지금 복잡도를 추가할 이유 없음(YAGNI — design-decision-discipline).
나중에 프로파일로 정말 병목이면, A-dedup 가 잎 코드/인터페이스 무수정으로 DeviceContext 쪽에만 dedup 을 더해 무손실 복원. 이는 D-RS-1(“DeviceContext = GL state 권위 + dirty-check”)의 기존 방향과 일관.
A→A-dedup 전이는 비파괴(잎/Processor/인터페이스 불변) → "seam 먼저, 최적화 나중" 정통.
즉 제 권고를 더 정확히 하면: §0.5 그대로(A) 가되, 그 안에서 "지금은 A-naive, 배치는 DeviceContext dedup(A-dedup)으로 나중에 무손실 추가" 입니다.

#include <vector>
#include "render/i_renderable.h"
namespace SJH::Scene { class Camera; }
namespace SJH
{
    class ICommandRecorder;

    // §0.5 Flat - mesh 특수 0, IRenderable* 만. (구 MeshPassProcessor + DrawCommand.WorldMesh)
    class RenderableProcessor
    {
      public:
        // collect 시점에 viewDepth 동봉 - depth 키 계산 위치 해결(§11): 카메라 아는 WorldPass 가 도출.
        void Submit(const IRenderable *r, float viewDepth)
        {
            mItems.push_back({r, r->QueueLayer(), viewDepth});   // queueLayer/depth 캡처(정렬 안정성)
        }
        void Clear() { mItems.clear(); }
        std::size_t Size() const { return mItems.size(); }

        // queueLayer asc -> (투명 queue>=2500) depth back-to-front / (불투명) front-to-back.
        // std::stable_sort 유지 = z-fighting 깜빡임 차단 + 골든 결정성(현 정책 보존).
        void Sort();

        // §0.5 본체 - ROP 적용 후 각 항목 자가발행. (lastProg/lastMat 배치 상태 없음)
        void Process(ICommandRecorder &rec, const Scene::Camera &cam)
        {
            rec.InvalidateStateCache();                 // foreign-GL 진입 후 캐시 desync 차단(D-RS-2)
            for (const auto &e : mItems) {
                rec.ApplyRenderStateBlock(e.renderable->GetRenderStateBlock());
                e.renderable->Render(rec, cam);
            }
        }

      private:
        struct Entry { const IRenderable *renderable; int queueLayer; float depth; };
        std::vector<Entry> mItems;
    };
}

// ── src/render/renderable_processor.cpp ─────────────────────────────
#include "render/renderable_processor.h"
#include "material/pass.h"
#include <algorithm>
namespace SJH
{
    void RenderableProcessor::Sort()
    {
        std::stable_sort(mItems.begin(), mItems.end(),
            [](const Entry &a, const Entry &b) {
                if (a.queueLayer != b.queueLayer) return a.queueLayer < b.queueLayer;
                if (Pass::IsTransparentQueue(a.queueLayer)) return a.depth < b.depth; // back-to-front
                return a.depth > b.depth;                                             // front-to-back
            });
        // 주: 구 코드의 program/material 그룹핑 2차/3차 키는 A 에선 생략(배치를 안 하므로).

    }

                const glm::mat4 mv = cam.GetViewMatrix() * a.GetWorldMatrix();
                mProc.Submit(mr, mv[3].z);             // ★ depth = view-space z (카메라 아는 여기서 도출)

GPU optimization > Draw calls, state changes, and APIs

Source: https://github.com/godotengine/godot-doc/blob/master/tutorials/performance/gpu_optimization.md


Godot sends instructions to the GPU via a graphics API (Vulkan, OpenGL, OpenGL ES or WebGL). The communication and driver activity involved can be quite costly, especially in OpenGL, OpenGL ES and WebGL. If we can provide these instructions in a way that is preferred by the driver and GPU, we can greatly increase performance. Nearly every API command in OpenGL requires a certain amount of validation to make sure the GPU is in the correct state. Even seemingly simple commands can lead to a flurry of behind-the-scenes housekeeping. Therefore, the goal is to reduce these instructions to a bare minimum and group together similar objects as much as possible so they can be rendered together, or with the minimum number of these expensive state changes.



GPU optimization > Draw calls, state changes, and APIs > 2D batching

Source: https://github.com/godotengine/godot-doc/blob/master/tutorials/performance/gpu_optimization.md


In 2D, the costs of treating each item individually can be prohibitively high - there can easily be thousands of them on the screen. This is why 2D batching is used. Multiple similar items are grouped together and rendered in a batch, via a single draw call, rather than making a separate draw call for each item. In addition, this means state changes, material and texture changes can be kept to a minimum.



GPU optimization > Draw calls, state changes, and APIs > 3D batching

Source: https://github.com/godotengine/godot-doc/blob/master/tutorials/performance/gpu_optimization.md


In 3D, we still aim to minimize draw calls and state changes. However, it can be more difficult to batch together several objects into a single draw call. 3D meshes tend to comprise hundreds or thousands of triangles, and combining large meshes in real-time is prohibitively expensive. The costs of joining them quickly exceeds any benefits as the number of triangles grows per mesh. A much better alternative is to join meshes ahead of time (static meshes in relation to each other). This can be done by artists, or programmatically within Godot using an add-on. There is also a cost to batching together objects in 3D. Several objects rendered as one cannot be individually culled. An entire city that is off-screen will still be rendered if it is joined to a single blade of grass that is on screen. Thus, you should always take objects' locations and culling into account when attempting to batch 3D objects together. Despite this, the benefits of joining static objects often outweigh other considerations, especially for large numbers of distant or low-poly objects.

3. 정정된 추천 — A-dedup (= §0.5 flat + 하위 캐시 dedup)
레퍼런스가 가리키는 정답은: flat §0.5 루프를 쓰되, 중복 제거를 DeviceContext/Program 캐시 층에 둔다. 이게 곧 Unity SRP Batcher("material 안 바뀌면 state 변경 회피") + Godot UniformSetCacheRD 의 구조이고, 우리 D-RS-1 이 이미 그 방향입니다.


// 루프/잎은 §0.5 그대로 (flat, mesh 특수 0, LSP-3) — 변경 없음
void RenderableProcessor::Process(DeviceContext& rc, const Camera& cam) {
    rc.InvalidateStateCache();
    for (auto& e : mItems) { rc.ApplyRenderStateBlock(e.renderable->GetRenderStateBlock()); e.renderable->Render(rc, cam); }
}

// 배치는 DeviceContext 캐시 층이 흡수 (D-RS-1 의 mLast 패턴 확장):
void DeviceContext::ApplyRenderStateBlock(const Pass::RenderStateBlock& w) {
    if (mStateInitialized && mLast == w) return;        // ← 이미 존재 (D-RS-1)
    ...apply...; mLast = w;
}
void DeviceContext::UseProgram(const Program& p) {
    if (mStateInitialized && mBoundProgram == &p) return; // ← mBoundProgram 이미 보유, 가드만 추가
    glUseProgram(p.GetID()); mBoundProgram = &p;
}
// FrameBlock(view/proj)/sampler 도 동형: Program 이 "직전 쓴 값과 같으면 skip" (Godot uniform-set cache 류)
왜 A-dedup 인가:

§0.5/D5 의 "Flat·mesh 특수 0·LSP-3" 에 완전 충실 (루프·잎·인터페이스 안 두꺼워짐).
🔵 3엔진 정통 (SRP Batcher persistent + Godot uniform cache + 중복 skip).
D-RS-1 의 기존 방향 연장 — 새 개념이 아니라 이미 있는 mLast dedup 을 UseProgram/UBO 로 확장.
픽셀 0 회귀 + 성능 0 회귀 (dedup 이 배치 효과 복원).
B 도 유효하지만: B(프로세서가 정렬·command 빌드·last-state 추적)는 🔵 Unreal FMeshPassProcessor 그 자체 — 우리 클래스가 이름까지 따왔으니 정통은 정통. 다만 프로세서가 mesh 지식(DrawCommand)을 계속 들어 §0.5 의 "mesh 특수 0" 와 어긋나고 두꺼워집니다. A-dedup 이 같은 성능을 §0.5 모양으로 얻습니다.
